cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(ConnectionTraceStudio VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build with Visual Studio C++ tools on Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "Instrument all user-mode targets with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(ConnectionTrace SDK/ConnectionTrace.cpp)
target_link_libraries(ConnectionTrace PUBLIC setupapi fwpuclnt ws2_32 rpcrt4)
add_executable(ConnectionTraceStudio WIN32 App/Main.cpp)
target_link_libraries(ConnectionTraceStudio PRIVATE ConnectionTrace user32 gdi32 comctl32 shell32)
target_link_options(ConnectionTraceStudio PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(ConnectionTraceCLI App/Console.cpp)
target_link_libraries(ConnectionTraceCLI PRIVATE ConnectionTrace)
add_executable(ConnectionTraceExample Examples/capture.cpp)
target_link_libraries(ConnectionTraceExample PRIVATE ConnectionTrace)
add_executable(ConnectionTraceTests Tests/ProtocolTests.cpp)
target_link_libraries(ConnectionTraceTests PRIVATE ConnectionTrace)
enable_testing()
add_test(NAME ProtocolAndCorrelation COMMAND ConnectionTraceTests)
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque signed TitanConnectionTrace INF/CAT/SYS package")
set(PACKAGE_RESOURCE "")
if(DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 foreach(name TitanConnectionTrace.inf TitanConnectionTrace.cat TitanConnectionTrace.sys)
  if(NOT EXISTS "${DRIVER_PACKAGE_DIRECTORY}/${name}")
   message(FATAL_ERROR "Incomplete opaque driver package: ${name}")
  endif()
 endforeach()
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "101 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanConnectionTrace.inf\"\n102 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanConnectionTrace.cat\"\n103 RCDATA \"${DRIVER_PACKAGE_DIRECTORY}/TitanConnectionTrace.sys\"\n")
endif()
add_library(TraceSetupCore Installer/Setup.cpp)
target_link_libraries(TraceSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(ConnectionTraceSetup WIN32 Installer/Main.cpp ${PACKAGE_RESOURCE})
add_executable(ConnectionTraceSetupConsole Installer/Console.cpp ${PACKAGE_RESOURCE})
foreach(target ConnectionTraceSetup ConnectionTraceSetupConsole)
 target_link_libraries(${target} PRIVATE TraceSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
add_executable(ConnectionTracePackageTests Tests/PackageTests.cpp)
target_link_libraries(ConnectionTracePackageTests PRIVATE TraceSetupCore)
add_test(NAME PackageIdentityAndTrust COMMAND ConnectionTracePackageTests)
